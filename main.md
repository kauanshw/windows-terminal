COMANDO = O QUE

# ENTRAR E SAIR
1. cd <nome pasta> = entra em uma pasta 
2. cd .. = sair da pasta atual 

# VER INFORMAÇÕES 
3. dir = mostra tudo que tem em uma pasta 

# LIMPAR TERMINAL 
4. cls = limpa o terminal

# PASTAS
5. mkdir <nome da pasta> = criar uma pasta 
6. rmdir < nome da pasta > = deletar uma pasta 

# ARQUIVOS
7. type nul > <nome do arquivo> = criar um arquivo 
8. del <nome do arquivo> = deletar um arquivo
